from itertools import product
from fractions import Fraction as F
states=list(product((0,1),repeat=2))
def terminal(obs):
    groups={}
    for s in states:
        groups.setdefault(tuple(s[i] for i in obs),[]).append(s[0]^s[1])
    return sum((F(max(sum(v),len(v)-sum(v)),4) for v in groups.values()),F(0))
for obs in [(),(0,),(1,),(0,1)]:
    v=terminal(obs)
    print(str(obs)+": terminal="+str(v)+", net="+str(v-F(len(obs),10)))
assert terminal(())==terminal((0,))==terminal((1,))==F(1,2)
assert terminal((0,1))-F(2,10)==F(4,5)
print("myopic_stops=True; bundle_improvement=3/10")
