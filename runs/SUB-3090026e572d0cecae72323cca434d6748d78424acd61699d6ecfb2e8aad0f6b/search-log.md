```python
# Own analytical fixture: independent exact Boolean predicates, no geometry.
# Fixed order; False is a sound disqualifying witness. No cache or learning.
rows = [(False, True, True, True)] * 3 + [(True, True, True, True)]
def run(early):
    work = 0
    for candidate, row in enumerate(rows):
        ok = True
        for predicate in row:
            work += 1
            ok = ok and predicate
            if early and not ok:
                break
        if ok:
            # Both arms pay an identical complete final reference.
            work += len(row)
            return candidate, work, tuple(row)
    return None, work, None
full, early = run(False), run(True)
assert full[0] == early[0] == 3
assert full[2] == early[2] == (True, True, True, True)
assert (full[1], early[1]) == (20, 11)
print('fixed_candidates=4; predicates_each=4; cache=none; final_reference=4')
```
