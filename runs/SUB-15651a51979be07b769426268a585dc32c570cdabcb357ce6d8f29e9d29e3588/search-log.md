"""Own scalar/table algebra only; no EDA process, training, or empirical model fit."""
import itertools, json, math, random
rng = random.Random(8142026)
K = 4

def dot(x,y): return sum(a*b for a,b in zip(x,y))
def det3(a):
    return (a[0][0]*(a[1][1]*a[2][2]-a[1][2]*a[2][1])
           -a[0][1]*(a[1][0]*a[2][2]-a[1][2]*a[2][0])
           +a[0][2]*(a[1][0]*a[2][1]-a[1][1]*a[2][0]))

def forward(t,i,j):
    ua,ub,um,vm,ud,vd = [t[n*K:(n+1)*K] for n in range(6)]
    z = 0.28 + ua[i] + ub[j]
    m = um[i]*vm[j]; d = ud[i]*vd[j]
    c = z + m
    return c+d,c-d,c,d

def loss(t,i,j,y,w,transformed=False):
    ab,ba,c,d = forward(t,i,j)
    if not transformed:
        return w[0]*(ab-y[0])**2+w[1]*(ba-y[1])**2
    ec=c-(y[0]+y[1])/2; ed=d-(y[0]-y[1])/2
    return (w[0]+w[1])*(ec*ec+ed*ed)+2*(w[0]-w[1])*ec*ed

def grad(t,i,j,y,w,transformed=False):
    ab,ba,c,d = forward(t,i,j)
    if transformed:
        ec=c-(y[0]+y[1])/2; ed=d-(y[0]-y[1])/2
        gc=2*(w[0]+w[1])*ec+2*(w[0]-w[1])*ed
        gd=2*(w[0]+w[1])*ed+2*(w[0]-w[1])*ec
    else:
        ga=2*w[0]*(ab-y[0]); gb=2*w[1]*(ba-y[1])
        gc=ga+gb; gd=ga-gb
    g=[0.0]*(6*K)
    g[i]=gc; g[K+j]=gc
    g[2*K+i]=gc*t[3*K+j]; g[3*K+j]=gc*t[2*K+i]
    g[4*K+i]=gd*t[5*K+j]; g[5*K+j]=gd*t[4*K+i]
    return g

masks=[x for x in itertools.product((0,1),repeat=4) if x[2]<=x[0] and x[3]<=x[1]]
obs_count=sum(2**sum(x) for x in masks)
max_loss=max_grad=max_fd=max_swap=0.0
for n in range(500):
    t=[rng.uniform(-2,2) for _ in range(6*K)]
    i,j=rng.randrange(K),rng.randrange(K)
    y=[rng.uniform(-4,4),rng.uniform(-4,4)]
    for flags in itertools.product((0,1),repeat=2):
        w=[flags[0]*0.7,flags[1]*1.3]
        max_loss=max(max_loss,abs(loss(t,i,j,y,w)-loss(t,i,j,y,w,True)))
        gd=grad(t,i,j,y,w); gt=grad(t,i,j,y,w,True)
        max_grad=max(max_grad,max(abs(a-b) for a,b in zip(gd,gt)))
    w=[0.7,1.3]; gd=grad(t,i,j,y,w)
    for l in range(6*K):
        plus=t.copy(); minus=t.copy(); plus[l]+=1e-5;minus[l]-=1e-5
        fd=(loss(plus,i,j,y,w)-loss(minus,i,j,y,w))/(2e-5)
        max_fd=max(max_fd,abs(fd-gd[l]))
    ab,ba,c,d=forward(t,i,j)
    max_swap=max(max_swap,abs((c-d)-ba),abs((c+d)-ab))
rank1_minor_max=0.0; rank2_det_max=0.0
for _ in range(100):
    u=[[rng.uniform(-2,2) for _ in range(2)] for _ in range(3)]
    v=[[rng.uniform(-2,2) for _ in range(2)] for _ in range(3)]
    mat=[[dot(a,b) for b in v] for a in u]
    rank2_det_max=max(rank2_det_max,abs(det3(mat)))
    a=[x[0] for x in u]; b=[x[0] for x in v]
    minor=(a[0]*b[0])*(a[1]*b[1])-(a[0]*b[1])*(a[1]*b[0])
    rank1_minor_max=max(rank1_minor_max,abs(minor))
identity=[[int(i==j) for j in range(3)] for i in range(3)]
def tri(x): return max(0,x+1)-2*max(0,x)+max(0,x-1)
direct_witness=[[tri(i-j) for j in range(3)] for i in range(3)]
assert direct_witness==identity
assert len(masks)==9 and obs_count==49
assert max_loss<1e-10 and max_grad<1e-10 and max_fd<1e-6
assert rank2_det_max<1e-10 and det3(identity)==1

     'observation_submask_count':obs_count,'parameter_vectors':500,'masked_loss_cases':2000,

     'rank1_minor_max_abs':rank1_minor_max,'rank2_det3_max_abs':rank2_det_max,
     'identity3_det':det3(identity),'three_relu_direct_identity_witness':direct_witness==identity,'K4_rank1_generated_scalars':6*K,
     'K4_rank2_generated_scalars':2*K+4*K*2,'K4_direct_branch_scalars':2*K*K}
