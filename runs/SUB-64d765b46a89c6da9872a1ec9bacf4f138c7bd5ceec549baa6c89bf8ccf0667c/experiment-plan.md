```python
"""Own finite, dimensionless DAG arithmetic; not an EDA simulator."""
import itertools, json

def primitive(node, xs, merge):
    if node == 'u':
        a,s=xs[0]; return a+1,s+1
    if node == 'r':
        (a,s),(b,t)=xs
        return (max(a+2,b+1),max(s+1,t+1)) if merge else (a+b+2,s+t+1)
    if node == 'z':
        a,s=xs[0]; return a+s,s+1
    raise ValueError(node)

parents={'u':('x',),'r':('u','y'),'z':('r',)}
nodes=tuple(parents)

def serial(worlds,merge):
    out=[];calls=0
    for x,y in worlds:
        st={'x':x,'y':y}
        for v in nodes:
            st[v]=primitive(v,tuple(st[p] for p in parents[v]),merge);calls+=1
        out.append(tuple(st[v] for v in nodes))
    return out,calls

def cached(worlds,merge):
    cache={};out=[];calls=0
    for x,y in worlds:
        st={'x':x,'y':y}
        for v in nodes:
            key=(v,merge,tuple(st[p] for p in parents[v]))
            if key not in cache:
                cache[key]=primitive(v,key[2],merge);calls+=1
            st[v]=cache[key]
        out.append(tuple(st[v] for v in nodes))
    return out,calls

def partitioned(worlds,merge):
    # Group at each operation by complete exact input tuple; regroup outputs
    # implicitly through equal tuple values on successor operations.
    states=[{'x':x,'y':y} for x,y in worlds];calls=0;sizes={}
    for v in nodes:
        groups={}
        for i,st in enumerate(states):
            key=(v,merge,tuple(st[p] for p in parents[v]))
            groups.setdefault(key,[]).append(i)
        sizes[v]=len(groups)
        for key,ids in groups.items():
            value=primitive(v,key[2],merge);calls+=1
            for i in ids:states[i][v]=value
    return [tuple(st[v] for v in nodes) for st in states],calls,sizes

rows=[]
for n,merge in itertools.product((1,2,4),(False,True)):
    worlds=[((i, i+1),(10,10)) for i in range(n)]
    a,ca=serial(worlds,merge);b,cb=cached(worlds,merge);c,cc,parts=partitioned(worlds,merge)
    assert a==b==c and cb==cc
    rows.append({'n':n,'dominating_reconvergence':merge,'serial_calls':ca,'cache_calls':cb,'partition_calls':cc,'classes':parts,'output_equal':True})
# All ordered four-world lists from a tiny four-state source alphabet.
alphabet=[((i,s),(10,10)) for i,s in itertools.product((0,1),(1,2))]
checked=0
for worlds in itertools.product(alphabet,repeat=4):
    for merge in (False,True):
        a,_=serial(worlds,merge);b,cb=cached(worlds,merge);c,cc,_=partitioned(worlds,merge)
        assert a==b==c and cb==cc
        checked+=1
# Unsafe arrival-only coalescence with slew-dependent primitive.
a1,s1=5,1;a2,s2=5,2
assert a1==a2 and a1+s1 != a2+s2
# Unsafe rounded arrival coalescence at a timing threshold.
from fractions import Fraction as F
left,right,required=F(999,1000),F(1001,1000),F(1)
assert round(float(left),2)==round(float(right),2)
assert required-left>0 and required-right<0
print(json.dumps({'kind':'own_abstract_arithmetic_not_EDA','rows':rows,'exhaustive_cases':checked,'incomplete_key_counterexample':{'equal_arrival':5,'slews':[1,2],'next_arrivals':[6,7]},'rounding_counterexample':{'arrivals':['999/1000','1001/1000'],'required':'1','slacks':['1/1000','-1/1000']},'runtime_speedup_measured':False},indent=2))
```
