from fractions import Fraction as F
from itertools import product
states = tuple(product((0,1), repeat=2))
def solve(pool, actions, depth, mode, costs, setup, first=True):
    labels = [s[0]^s[1] if mode == "xor" else s[0] for s in pool]
    best = F(max(sum(labels),len(labels)-sum(labels)),len(labels))
    if depth:
        for a in actions:
            v = -costs[a] - (setup if first else 0)
            for bit in (0,1):
                g = tuple(s for s in pool if s[a] == bit)
                if g:
                    v += F(len(g),len(pool))*solve(g,tuple(i for i in actions if i!=a),depth-1,mode,costs,setup,False)
            best = max(best,v)
    return best
for mode in ("xor","single"):
    vals = [solve(states,(0,1),d,mode,(F(1,10),)*2,F(0)) for d in (1,2)]
    print(mode, *map(str,vals))
    assert vals == ([F(1,2),F(4,5)] if mode=="xor" else [F(9,10)]*2)
count=improved=0
for mode,a,b,s in product(("xor","single"),(F(1,20),F(1,10),F(1,5),F(3,10)),(F(1,20),F(1,10),F(1,5),F(3,10)),(F(0),F(1,10),F(1,4))):
    v1=solve(states,(0,1),1,mode,(a,b),s)
    v2=solve(states,(0,1),2,mode,(a,b),s)
    count+=1; improved+=v2>v1
print("grid",count,"depth2_improves",improved)
assert (count,improved)==(96,27)
