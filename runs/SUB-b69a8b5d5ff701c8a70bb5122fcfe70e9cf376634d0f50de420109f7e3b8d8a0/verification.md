from itertools import product
from fractions import Fraction

def accuracy(n, observed):
    groups = {}
    for s in product((0, 1), repeat=n):
        key = tuple(s[i] for i in observed)
        groups.setdefault(key, [0, 0])[sum(s) % 2] += 1
    return sum((Fraction(max(v), 2**n) for v in groups.values()), Fraction())
for n, obs in [(2, (0,)), (2, (0,1)), (3, (0,1)), (3, (0,1,2))]:
